# Dependencies for Salah App Sentiment Analysis
# See implementation_plan.md §14

# ─── Core ────────────────────────────────────────────────────────────────────────
pandas
numpy
pyarrow

# ─── NLP / Sentiment ─────────────────────────────────────────────────────────────
nltk                      # VADER
transformers              # RoBERTa, BART-MNLI
torch                     # PyTorch backend

bertopic                  # Topic modelling

pingouin                  # Effect sizes, bootstrap CIs

matplotlib
